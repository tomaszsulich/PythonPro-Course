# cars/admin.py
from django.contrib import admin
from django.http import HttpRequest

from .models import Car


@admin.register(Car)
class CarAdmin(admin.ModelAdmin):
    list_display = (
        "brand",
        "model",
        "full_name",
        "year",
        "is_available",
    )

    search_fields = (
        "brand",
        "model",
    )

    list_filter = (
        "is_available",
        "year",
    )

    ordering = (
        "-year",
    )

    actions = (
        "mark_as_unavailable",
        "mark_as_available",
    )

    def get_readonly_fields(self, request: HttpRequest, obj: Car | None = None) -> tuple[str, ...]:
        if obj is not None:
            return ("year",)

        return ()

    def full_name(self, obj: Car) -> str:
        return f"{obj.brand} {obj.model}"

    full_name.short_description = "Pełna nazwa"

    def mark_as_unavailable(self, request: HttpRequest, queryset) -> None:
        updated = queryset.update(is_available=False)

        self.message_user(
            request,
            f"Gotowe! Liczba samochodów wycofanych z oferty: {updated}. "
            "Miejmy nadzieję, że klienci jeszcze nie są w drodze😅"
        )

    mark_as_unavailable.short_description = "Oznacz jako niedostępne"

    def mark_as_available(self, request: HttpRequest, queryset) -> None:
        updated = queryset.update(is_available=True)

        self.message_user(
            request,
            f"Gotowe! Liczba samochodów przywróconych do oferty: {updated}.",
        )

    mark_as_available.short_description = "Oznacz jako dostępne"


# Uruchomienie serwera
python manage.py runserver
