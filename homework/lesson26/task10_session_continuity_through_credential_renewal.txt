# Odświeżenie access tokena przez Postman

Method: POST
URL: http://127.0.0.1:8000/auth/jwt/refresh/

# Body (raw/JSON)

{
    "refresh": "<refresh_token>"
}


# Otrzymany nowy access token zostaje wykorzystany
# do ponownego uwierzytelnienia w chronionym endpoincie.

Method: GET
URL: http://127.0.0.1:8000/api/protected/

Authorization:
Bearer <new_access_token>
