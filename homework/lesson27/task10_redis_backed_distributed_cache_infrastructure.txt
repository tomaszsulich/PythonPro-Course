# Instalacja biblioteki django-redis
pip install django-redis


# Aktualizacja pliku requirements.txt
pip freeze > requirements.txt


# Uruchomienie Redis w kontenerze Docker
docker run -d --name redis-cache -p 6379:6379 redis


# Sprawdzenie działania kontenera
docker ps

OUTPUT:

dbc5b6a24256   redis     "docker-entrypoint.s…"   17 seconds ago   Up 16 seconds   0.0.0.0:6379->6379/tcp, [::]:6379->6379/tcp   redis-cache


# config/settings.py
CACHES = {
    "default": {
        "BACKEND": "django_redis.cache.RedisCache",
        "LOCATION": "redis://127.0.0.1:6379/1",
        "OPTIONS": {
            "CLIENT_CLASS": "django_redis.client.DefaultClient",
        },
    }
}


# Uruchomienie serwera django
python manage.py runserver
