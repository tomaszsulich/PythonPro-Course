# cache_lab/views.py
from django.http import HttpRequest, HttpResponse
from django.shortcuts import render
from django.views.decorators.cache import cache_page

from rest_framework.decorators import api_view
from rest_framework.request import Request
from rest_framework.response import Response


def cache_test(request: HttpRequest) -> HttpResponse:
    return render(
        request,
        "cache_lab/cache_test.html",
    )


@cache_page(60)
@api_view(["GET"])
def product_list(request: Request):
    return Response(
        {
            "products": [
                {"id": 1, "name": "Kanapka z jajkiem"},
                {"id": 2, "name": "Tost z szynką i serem"},
            ]
        }
    )


# cache_lab/urls.py
from django.urls import path

from .views import cache_test, product_list


urlpatterns = [
    path("cache-test/", cache_test, name="cache-test"),
    path("products/", product_list, name="product-list"),
]


# config/urls.py
from django.contrib import admin

urlpatterns = [
